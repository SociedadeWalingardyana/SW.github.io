let userCountry = "Desconhecido"; // Pega o país do usuário via IP fetch("https://ipapi.co/json/") .then(res => res.json()) .then(data => { userCountry = data.country_name || "Desconhecido"; }); // Função para criar comentário com país function addComment(text, country) { const p = document.createElement("p"); p.textContent = `• ${text} - ${country}`; section.appendChild(p); } // Enviar comentário btnComment.onclick = () => { if(!textarea.value.trim()) return; db.ref("comentarios").push({texto: textarea.value, data: Date.now(), pais: userCountry}); addComment(textarea.value, userCountry); textarea.value = ""; }; // Receber comentários do banco e exibir com país db.ref("comentarios").limitToLast(20).once("value", snap=>{ snap.forEach(c => { addComment(c.val().texto, c.val().pais || "Desconhecido"); }); }); db.ref("comentarios").limitToLast(1).on("child_added", snap=>{ addComment(snap.val().texto, snap.val().pais || "Desconhecido"); });
