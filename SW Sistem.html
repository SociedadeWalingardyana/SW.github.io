/* Carregar histórico de vendas e calcular finanças */ function loadSales(){ const ul = document.getElementById('sales'); ul.innerHTML = ''; const vendas = JSON.parse(localStorage.getItem('vendas')) || []; let totalGanhos = 0; // Criar cabeçalho de resumo const resumo = document.createElement('div'); resumo.style.marginBottom = "20px"; resumo.style.padding = "10px"; resumo.style.border = "1px solid #333"; resumo.style.borderRadius = "8px"; vendas.forEach(s => { const li = document.createElement('li'); li.style.padding = "8px 0"; li.style.borderBottom = "1px solid #222"; li.innerHTML = `R$ ${s.valor.toFixed(2)} - ${s.book}
${s.data}`; ul.appendChild(li); totalGanhos += s.valor; }); const lucroNeto = totalGanhos * 0.95; // Exemplo: descontando 5% de taxas do Mercado Pago resumo.innerHTML = `

Vendas Totais: ${vendas.length}

Ganhos Brutos: R$ ${totalGanhos.toFixed(2)}

Lucro Estimado (Líquido): R$ ${lucroNeto.toFixed(2)}

`; ul.prepend(resumo); }
